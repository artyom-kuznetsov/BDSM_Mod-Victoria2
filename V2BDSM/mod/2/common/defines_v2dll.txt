# V2DLL settings tied to this mod's game data. Created automatically, edit freely.
# Every player in a multiplayer game must have identical values here, otherwise OOS.
# '#' starts a comment.

# --- Factories (PATCH_FACTORY_AUTO_CLOSE_UNPROFITABLE in v2dll_settings.ini) ---
# A factory that is NOT subsidized and has been unprofitable (sales below the cost of its
# input goods) for this many days in a row is closed automatically, as if closed by hand.
# The money stored in the factory is paid out to the capitalists of its state.
# 0 = never close automatically.
factory_unprofitable_close_days = 60

# --- Minting (ENABLE_MINTING in v2dll_settings.ini) ---
# Daily income every country gets. Added to the treasury once per day, shown in the budget
# window as "minting_inc", included in total income (budget window, topbar chart, tooltips).
#
# minting_formula - evaluated per country, result is pounds per day. Empty = minting is off.
# Variables:
#   industry_score     industrial score of the country (same as industrial_score in scripts)
#   total_population   country population (the number the game stores for the country)
#   any other name     a country variable declared in the game (set_variable = { which = NAME ... }),
#                      e.g. economic_thought_level; its value, or 0 if the country does not have it
# Operators: + - * /   parentheses, decimal numbers (use a dot: 0.01)
minting_formula = industry_score + industry_score * monetary_system_level * 0.25

# --- AI factory expansion (PATCH_AI_EXPAND_STAFFING in v2dll_settings.ini) ---
# An AI country expands an existing factory only if at least this percent of the factory's
# jobs are filled with workers (employees / (workforce * level)). The vanilla AI only looks at
# the unemployed share of the whole state and builds the most profitable factory type up to
# huge levels with almost no workers. Capitalists already need about 90 percent in vanilla.
# 0 = no limit (vanilla AI behaviour).
ai_factory_expand_min_staffing = 90

# --- Factory minimum wage (PATCH_FACTORY_MIN_WAGE in v2dll_settings.ini) ---
# Every day a factory pays its workers at least this many pounds per 10000 workers (owners and
# slaves are not counted), whether or not the factory is profitable. The wage is paid only out of
# the money the factory has (its budget): if the factory has less, it pays what it has, and its
# balance never goes below zero. Where the profit allows a higher wage, the game's own logic
# (the 'leftover' share and the minimum_wage reform) applies as usual: the larger of the two is
# paid. For a subsidized factory the wage floor is added to its subsidy (the state pays it),
# so that the factory can still buy all its input goods.
# Fractions are allowed (use a dot). 0 = off (the game's own minimum wage only).
factory_min_wage_per_10000 = 9

# --- AI naval bases (PATCH_AI_NAVAL_BASE_LIMIT in v2dll_settings.ini) ---
# In how many provinces of one state an AI country may have a naval base. Once a state has
# that many provinces with a naval base (built or being built), the AI does not start a base
# in another province of the state. Upgrading a base the AI already has is not limited. The
# player and the capitalists are not affected. 0 = no limit (vanilla AI behaviour).
ai_naval_base_max_provinces_per_state = 1

# --- Pop salary bonus (POP_SALARY_ADDITIONAL_MULTIPLIER in v2dll_settings.ini) ---
# Every income a pop receives (factory wages, RGO and artisan income, owners' profit share,
# budget payouts to clergy/bureaucrats/soldiers) is increased by this many percent AFTER the
# income tax is taken, so the pops keep the extra money to spend. The extra money is created
# out of nothing: the employer, the factory and the treasury do not pay it, taxes are
# not charged on it. A formula like minting_formula (same variables and operators), evaluated
# per country once per game month, so it can differ by country and change during the game.
# The national modifier pop_income_modifier (technologies, inventions, events, ...; 0.1 = +10)
# is added to the result. Result in percent, 0..1000 (negative = 0). 0 = no formula bonus.
pop_salary_additional_multiplier = 0
